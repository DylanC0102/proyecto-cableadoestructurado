  <script>
    const DEMO_DATA = {
      version: "v1.0.0",
      fechaActualizacion: new Date().toLocaleDateString() + ' ' + new Date().toLocaleTimeString(),
      ordenU: "mayor_arriba",
      vistaActual: "frontal",
      rackSeleccionadoId: "R01",
      racks: [
        { id: "R01", nombre: "Rack demostrativo 01", totalU: 18, ubicacion: "Cuarto Principal CT-01" },
        { id: "R02", nombre: "Rack demostrativo 02", totalU: 12, ubicacion: "Cuarto Secundario CT-02" }
      ],
      elementos: [
        { id: "EQ-001", rackId: "R01", tipo: "equipo", nombre: "Patch panel cobre", subtipo: "Patch panel", uInicio: 18, uAlto: 1, estado: "Confirmado" },
        { id: "EQ-002", rackId: "R01", tipo: "equipo", nombre: "Organizador horizontal", subtipo: "Organizador", uInicio: 17, uAlto: 1, estado: "Confirmado" },
        { id: "EQ-003", rackId: "R01", tipo: "equipo", nombre: "Switch de acceso", subtipo: "Switch", uInicio: 16, uAlto: 1, estado: "Inferido", circuito: "CIR-A [POR VALIDAR]" },
        { id: "RES-001", rackId: "R01", tipo: "reserva", nombre: "RESERVADO - Crecimiento [EJEMPLO]", uInicio: 11, uAlto: 3, estado: "Ejemplo" },
        { id: "EQ-004", rackId: "R01", tipo: "equipo", nombre: "UPS de rack", subtipo: "UPS", uInicio: 2, uAlto: 2, estado: "Confirmado" },
        { id: "EQ-005", rackId: "R02", tipo: "equipo", nombre: "Panel de fibra", subtipo: "Patch panel", uInicio: 12, uAlto: 1, estado: "Pendiente" }
      ],
      bitacora: [
        { timestamp: new Date().toLocaleTimeString(), accion: "Inicialización del modelo verificado tres capas" }
      ]
    };

    let app = JSON.parse(localStorage.getItem('RACK_DASHBOARD_DATA')) || DEMO_DATA;

    document.addEventListener('DOMContentLoaded', () => { renderizarTodo(); });

    function guardarEstado() {
      localStorage.setItem('RACK_DASHBOARD_DATA', JSON.stringify(app));
      renderizarTodo();
    }

    function registrarBitacora(accion) {
      app.bitacora.unshift({ timestamp: new Date().toLocaleTimeString(), accion });
      if(app.bitacora.length > 6) app.bitacora.pop();
    }

    function renderizarTodo() {
      document.getElementById('lbl-fecha').textContent = app.fechaActualizacion;
      renderizarKPIs();
      renderizarListaRacks();
      renderizarRackGrid();
      renderizarTablaInventario();
      renderizarAlertas();
      renderizarBitacora();
    }

    function renderizarKPIs() {
      document.getElementById('kpi-racks').textContent = app.racks.length;
      let totalUDisponibles = app.racks.reduce((acc, r) => acc + r.totalU, 0);
      let totalUOcupadas = app.elementos.reduce((acc, el) => acc + el.uAlto, 0);
      let porcentaje = totalUDisponibles > 0 ? Math.round((totalUOcupadas / totalUDisponibles) * 100) : 0;
      
      document.getElementById('kpi-ocupacion').textContent = porcentaje + '%';
      document.getElementById('kpi-u-libres').textContent = totalUDisponibles - totalUOcupadas;
      
      let pendientes = app.elementos.filter(el => el.estado === 'Pendiente' || el.estado === 'Inferido').length;
      document.getElementById('kpi-pendientes').textContent = pendientes;
    }

    function renderizarListaRacks() {
      const container = document.getElementById('racks-list');
      container.innerHTML = '';
      app.racks.forEach(r => {
        const div = document.createElement('div');
        div.className = `rack-item ${r.id === app.rackSeleccionadoId ? 'active' : ''}`;
        div.onclick = () => { app.rackSeleccionadoId = r.id; guardarEstado(); };
        div.innerHTML = `<div class="rack-item-title">${r.nombre}</div><div class="rack-item-sub">${r.totalU}U | ${r.ubicacion}</div>`;
        container.appendChild(div);
      });
    }

    function renderizarRackGrid() {
      const rack = app.racks.find(r => r.id === app.rackSeleccionadoId);
      const container = document.getElementById('rack-grid-container');
      container.innerHTML = '';
      if(!rack) return;

      document.getElementById('rack-active-title').textContent = rack.nombre;

      const ocupacion = {};
      app.elementos.filter(e => e.rackId === rack.id).forEach(el => {
        for(let u = el.uInicio; u < el.uInicio + el.uAlto; u++) ocupacion[u] = el;
      });

      let secuencia = [];
      for(let u = 1; u <= rack.totalU; u++) secuencia.push(u);
      if(app.ordenU === 'mayor_arriba') secuencia.reverse();

      secuencia.forEach(u => {
        const row = document.createElement('div');
        row.className = 'rack-u-row';
        const item = ocupacion[u];

        let contentHtml = `<div class="rack-u-content" style="color: var(--text-muted);">Espacio Libre</div>`;
        if(item) {
          const isReserva = item.tipo === 'reserva';
          const claseBg = isReserva ? 'u-reserved' : 'u-occupied';
          contentHtml = `
            <div class="rack-u-content ${claseBg}" onclick="mostrarFicha('${item.id}')" style="cursor:pointer;">
              <span>${item.nombre}</span>
              <span class="tag tag-${item.estado.toLowerCase()}">${item.estado}</span>
            </div>
          `;
        }
        row.innerHTML = `<div class="rack-u-num">${u}</div>${contentHtml}<div class="rack-u-num">${u}</div>`;
        container.appendChild(row);
      });
    }

    function renderizarTablaInventario() {
      const tbody = document.getElementById('tbl-inventario-body');
      tbody.innerHTML = '';

      app.elementos.forEach(el => {
        const tr = document.createElement('tr');
        tr.innerHTML = `
          <td><strong>${el.id}</strong></td>
          <td>${el.nombre}</td>
          <td>${el.rackId} / U${el.uInicio}</td>
          <td><span class="tag tag-${el.estado.toLowerCase()}">${el.estado}</span></td>
          <td>
            <button class="btn btn-danger" style="padding: 2px 6px; font-size: 0.7rem;" onclick="eliminarElemento('${el.id}')">Eliminar</button>
          </td>
        `;
        tbody.appendChild(tr);
      });
    }

    function mostrarFicha(id) {
      const el = app.elementos.find(e => e.id === id);
      if(!el) return;
      document.getElementById('ficha-detalle').innerHTML = `
        <div style="font-weight:bold; color:#fff; margin-bottom:4px;">${el.nombre}</div>
        <div><strong>Capa Física:</strong> Rack ${el.rackId} - U${el.uInicio} (${el.uAlto}U)</div>
        <div><strong>Capa Lógica:</strong> ${el.subtipo || 'Reserva de Espacio'}</div>
        <div><strong>Capa Eléctrica:</strong> ${el.circuito || 'Rama A/B [Estándar]'}</div>
        <div><strong>Estado del Dato:</strong> <span class="tag tag-${el.estado.toLowerCase()}">${el.estado}</span></div>
      `;
    }

    function renderizarAlertas() {
      const container = document.getElementById('panel-alertas');
      container.innerHTML = '';
      const alertas = app.elementos.filter(el => el.estado === 'Pendiente' || el.estado === 'Inferido');
      if(alertas.length === 0) {
        container.innerHTML = `<div style="color:var(--text-muted); font-size:0.8rem;">Sin alertas de datos no comprobados.</div>`;
        return;
      }
      alertas.forEach(a => {
        const div = document.createElement('div');
        div.className = 'alert-item';
        div.textContent = `${a.id}: ${a.nombre} - [${a.estado}] Requiere verificación en sitio.`;
        container.appendChild(div);
      });
    }

    function renderizarBitacora() {
      const container = document.getElementById('panel-bitacora');
      container.innerHTML = '';
      app.bitacora.forEach(b => {
        const div = document.createElement('div');
        div.className = 'bitacora-item';
        div.innerHTML = `<strong>${b.timestamp}:</strong> ${b.accion}`;
        container.appendChild(div);
      });
    }

    function alternarModoForm(modo) {
      document.getElementById('group-tipo').style.display = modo === 'reserva' ? 'none' : 'block';
    }

    function guardarElemento(e) {
      e.preventDefault();
      const modo = document.getElementById('reg-modo').value;
      const nombre = document.getElementById('reg-nombre').value;
      const tipo = document.getElementById('reg-tipo').value;
      const uInicio = parseInt(document.getElementById('reg-u-inicio').value);
      const uAlto = parseInt(document.getElementById('reg-u-alto').value);
      const estado = document.getElementById('reg-estado').value;

      const rack = app.racks.find(r => r.id === app.rackSeleccionadoId);

      if(uInicio + uAlto - 1 > rack.totalU) {
        alert(`Error de capacidad: El elemento excede las ${rack.totalU}U disponibles.`);
        return;
      }

      // Validar solapamiento
      const solapa = app.elementos.some(el => {
        if(el.rackId !== app.rackSeleccionadoId) return false;
        return (uInicio <= (el.uInicio + el.uAlto - 1) && (uInicio + uAlto - 1) >= el.uInicio);
      });

      if(solapa) {
        alert("Error de conflicto: Ya existe un equipo o reserva asignado a esa posición U.");
        return;
      }

      const nuevoId = (modo === 'equipo' ? 'EQ-' : 'RES-') + String(Date.now()).slice(-3);
      app.elementos.push({
        id: nuevoId,
        rackId: app.rackSeleccionadoId,
        tipo: modo,
        nombre: nombre,
        subtipo: tipo,
        uInicio: uInicio,
        uAlto: uAlto,
        estado: estado
      });

      registrarBitacora(`Elemento ${nuevoId} registrado en ${app.rackSeleccionadoId}`);
      guardarEstado();
      document.getElementById('form-registro').reset();
    }

    function eliminarElemento(id) {
      if(!confirm(`¿Desea eliminar el elemento ${id}?`)) return;
      app.elementos = app.elementos.filter(e => e.id !== id);
      registrarBitacora(`Elemento ${id} eliminado`);
      guardarEstado();
    }

    function cambiarOrdenU(orden) { app.ordenU = orden; guardarEstado(); }
    function cambiarVista(vista) { app.vistaActual = vista; renderizarRackGrid(); }

    function exportarJSON() {
      const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(app, null, 2));
      const anchor = document.createElement('a');
      anchor.setAttribute("href", dataStr);
      anchor.setAttribute("download", `inventario-respaldo.json`);
      document.body.appendChild(anchor);
      anchor.click();
      anchor.remove();
      registrarBitacora("Respaldo JSON exportado");
    }

    function importarJSON(event) {
      const file = event.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function(e) {
        try {
          const jsonImportado = JSON.parse(e.target.result);
          if (jsonImportado.racks && jsonImportado.elementos) {
            app = jsonImportado;
            registrarBitacora("Respaldo JSON importado con éxito");
            guardarEstado();
          } else {
            alert("Estructura JSON inválida.");
